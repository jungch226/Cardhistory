async function previewImage(event) { const file = event.target.files?.[0]; if (!file) return; const MAX_SIZE = 1600; const QUALITY = 0.8; try { const bitmap = await createImageBitmap(file, { imageOrientation: "from-image" }); const scale = Math.min( 1, MAX_SIZE / Math.max(bitmap.width, bitmap.height) ); const width = Math.round(bitmap.width * scale); const height = Math.round(bitmap.height * scale); const canvas = document.createElement("canvas"); canvas.width = width; canvas.height = height; const ctx = canvas.getContext("2d"); ctx.fillStyle = "#FFFFFF"; ctx.fillRect(0, 0, width, height); ctx.drawImage(bitmap, 0, 0, width, height); bitmap.close(); base64ImageData = canvas.toDataURL( "image/jpeg", QUALITY ); const preview = document.getElementById("preview"); preview.src = base64ImageData; preview.style.display = "block"; document.getElementById("previewText").style.display = "none"; const originalKB = (file.size / 1024).toFixed(0); const compressedKB = ( base64ImageData.length * 0.75 / 1024 ).toFixed(0); console.log( `사진 압축: ${originalKB}KB → ${compressedKB}KB` ); canvas.width = 0; canvas.height = 0; } catch (error) { console.error("사진 처리 실패:", error); alert("사진을 처리하지 못했습니다."); } }
